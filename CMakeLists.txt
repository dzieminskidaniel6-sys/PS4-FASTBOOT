cmake_minimum_required(VERSION 3.16)
project(ps4_fastboot_flasher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(fastboot_protocol src/fastboot_protocol.cpp)
target_include_directories(fastboot_protocol PUBLIC src)

option(BUILD_TESTING "Build host-side tests" ON)
if(BUILD_TESTING)
  enable_testing()
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE fastboot_protocol)
  add_test(NAME protocol_tests COMMAND protocol_tests)
endif()
